## HashMap
```java
import java.util.HashMap;
import java.util.Map;

public class HashMapExample {
    public static void main(String[] args) {
        // 1. 创建 HashMap
        Map<String, Integer> map = new HashMap<>();

        // 2. 添加元素 put(key, value)
        map.put("Alice", 25);
        map.put("Bob", 30);
        map.put("Charlie", 35);

        // 3. 获取元素 get(key)
        System.out.println("Alice的年龄: " + map.get("Alice")); // 25

        // 4. 判断 key 是否存在 containsKey(key)
        System.out.println("是否包含Bob: " + map.containsKey("Bob")); // true

        // 5. 判断 value 是否存在 containsValue(value)
        System.out.println("是否有年龄为30的人: " + map.containsValue(30)); // true

        // 6. 删除元素 remove(key)
        map.remove("Charlie");
        System.out.println("删除Charlie后: " + map);

        // 7. 遍历 keySet
        System.out.println("遍历key:");
        for (String key : map.keySet()) {
            System.out.println(key);
        }

        // 8. 遍历 values
        System.out.println("遍历value:");
        for (Integer value : map.values()) {
            System.out.println(value);
        }

        // 9. 遍历 entrySet
        System.out.println("遍历键值对:");
        for (Map.Entry<String, Integer> entry : map.entrySet()) {
            System.out.println(entry.getKey() + " -> " + entry.getValue());
        }

        // 10. 获取大小 size
        System.out.println("Map大小: " + map.size());

        // 11. 清空 clear
        map.clear();
        System.out.println("清空后Map: " + map);
    }
}

```

```java
class Solution {
    int max = Integer.MIN_VALUE;
    public int maxPathSum(TreeNode root) {
        if(root == null) return 0;
        dfs(root);
        return max;
    }

    public int dfs(TreeNode root) {
        if(root == null) return 0;
        int lenum = Math.max(dfs(root.left), 0);
        int rinum = Math.max(dfs(root.right), 0);
        max= Math.max(max, lenum + rinum + root.val);
        return root.val + Math.max(lenum, rinum);
    }
}
```

```java
class Solution {
    public ListNode reverseList(ListNode head) {
        ListNode next = head;			//主节点
        ListNode result = null;			//结果节点
        while(next != null) {
            ListNode temp = next.next;   //暂为主右
            next.next = result;			 //主指结
            result = next;				 //结为主
            next = temp;				 //主为暂
        }
        return result;
    }
}
```

```java
public class Solution {
    public boolean hasCycle(ListNode head) {
        if(head == null) return false;
        ListNode slow = head;
        ListNode fast = head;
        while(fast != null && fast.next != null) {	//注意条件
            slow = slow.next;
            fast = fast.next.next;
            if(slow == fast) return true;	//最后判断
        }
        return false;
    }
}
```

```java
class Solution {
    public ListNode addTwoNumbers(ListNode l1, ListNode l2) {
        ListNode node = new ListNode();
        ListNode res = node;
        int carry = 0;
        while(l1 != null || l2 != null || carry != 0) {
            int n1 = l1 == null ? 0 : l1.val;
            int n2 = l2 == null ? 0 : l2.val;
            int temp = n1 + n2 + carry;
            carry = temp /10;
            node.next = new ListNode(temp % 10);
            node = node.next;
            if(l1 != null) l1 = l1.next;
            if(l2 != null) l2 = l2.next;
        }
        return res.next;
    }
}
```

### 双向链表
```java
public class DoublyLinkedList<T> {

    // 节点定义
    static class Node<T> {
        T value;
        Node<T> prev;
        Node<T> next;

        Node(T value) {
            this.value = value;
        }
    }

    private Node<T> head; // 虚拟头
    private Node<T> tail; // 虚拟尾
    private int size;

    public DoublyLinkedList() {
        head = new Node<>(null);
        tail = new Node<>(null);
        head.next = tail;
        tail.prev = head;
        size = 0;
    }

    // 在头部插入节点
    public void addFirst(T value) {
        Node<T> node = new Node<>(value);
        node.next = head.next;
        node.prev = head;
        head.next.prev = node;
        head.next = node;
        size++;
    }

    // 在尾部插入节点
    public void addLast(T value) {
        Node<T> node = new Node<>(value);
        node.prev = tail.prev;
        node.next = tail;
        tail.prev.next = node;
        tail.prev = node;
        size++;
    }

    // 删除指定节点
    public void remove(Node<T> node) {
        if (node == null || node == head || node == tail) return;
        node.prev.next = node.next;
        node.next.prev = node.prev;
        size--;
    }

    // 删除尾节点并返回
    public Node<T> removeLast() {
        if (tail.prev == head) return null; // 空链表
        Node<T> last = tail.prev;
        remove(last);
        return last;
    }

    // 获取链表长度
    public int getSize() {
        return size;
    }

    // 遍历打印（调试用）
    public void printList() {
        Node<T> curr = head.next;
        while (curr != tail) {
            System.out.print(curr.value + " ");
            curr = curr.next;
        }
        System.out.println();
    }
}
```

```java
class LRUCache {

    class DoubleLinkedNode{
        int key;
        int value;
        DoubleLinkedNode prev;
        DoubleLinkedNode next;

        public DoubleLinkedNode() {}

        DoubleLinkedNode(int key, int value) {
            this.key = key;
            this.value = value;
        }
    }

    DoubleLinkedNode head;
    DoubleLinkedNode tail;
    int capacity;
    int size;
    Map<Integer, DoubleLinkedNode> mymap = new HashMap<>();

    public LRUCache(int capacity) { //初始化
        this.size = 0;
        this.capacity = capacity;
        head = new DoubleLinkedNode();
        tail = new DoubleLinkedNode();
        head.next = tail;
        tail.prev = head;
    }
    
    public int get(int key) {   //查找缓存
        DoubleLinkedNode node = mymap.get(key);
        if(node == null) return -1;
        moveToHead(node);  //查找到，移动到头
        return node.value;
    }
    
    public void put(int key, int value) {   //插入
        DoubleLinkedNode node = mymap.get(key);
        if(node == null) {
            //不存在
            DoubleLinkedNode newNode = new DoubleLinkedNode(key, value);
            mymap.put(key, newNode);
            addToHead(newNode);   //新节点添加到头
            size++;
            if(size > capacity) {
                DoubleLinkedNode tailNode = removeTail();   //删除尾
                mymap.remove(tailNode.key);
                size--;
            }
        } else {
            //存在
            node.value = value;
            moveToHead(node);   //移动到头
        }
    }

    ////////////////////////

    
    private void addToHead(DoubleLinkedNode node) {
        node.prev = head;
        node.next = head.next;
        head.next.prev = node;
        head.next = node;
    }

    private void removeNode(DoubleLinkedNode node) {
        node.prev.next = node.next;
        node.next.prev = node.prev;
    }

    private void moveToHead(DoubleLinkedNode node) {
        removeNode(node);
        addToHead(node);
    }

    private DoubleLinkedNode removeTail() {
        DoubleLinkedNode res = tail.prev;
        removeNode(res);
        return res;
    }
}

 * LRUCache obj = new LRUCache(capacity);
 * int param_1 = obj.get(key);
 * obj.put(key,value);
 */
```

## 栈
```java
import java.util.ArrayDeque;
import java.util.Deque;

public class StackDemo {
    public static void main(String[] args) {
        // 创建栈（使用 ArrayDeque）
        Deque<Integer> stack = new ArrayDeque<>();

        // 1. 入栈 push
        stack.push(10);
        stack.push(20);
        stack.push(30);
        System.out.println("栈内容: " + stack); // 输出类似: [30, 20, 10]

        // 2. 查看栈顶元素 peek（不删除）
        System.out.println("栈顶元素: " + stack.peek()); // 输出: 30

        // 3. 出栈 pop（删除栈顶元素）
        int popped = stack.pop();
        System.out.println("弹出的元素: " + popped); // 输出: 30
        System.out.println("栈内容: " + stack);      // 输出类似: [20, 10]

        // 4. 判断栈是否为空 isEmpty
        System.out.println("栈是否为空: " + stack.isEmpty()); // 输出: false

        // 5. 栈大小 size
        System.out.println("栈大小: " + stack.size()); // 输出: 2

        // 6. 模拟 search（从栈顶开始的距离）
        System.out.println("20 在栈顶距离: " + searchFromTop(stack, 20));   // 输出: 1
        System.out.println("100 在栈顶距离: " + searchFromTop(stack, 100)); // 输出: -1

        // 遍历栈（Deque 默认从栈顶到栈底）
        System.out.print("栈遍历: ");
        for (int num : stack) {
            System.out.print(num + " ");
        }
        System.out.println();
    }

    // 模拟 Stack.search 的功能
    public static <T> int searchFromTop(Deque<T> stack, T target) {
        int distance = 1; // 从栈顶开始计数
        for (T item : stack) {
            if (item.equals(target)) {
                return distance;
            }
            distance++;
        }
        return -1;
    }
}
```

## 队列
```java
import java.util.ArrayDeque;
import java.util.Queue;

public class QueueDemo {
    public static void main(String[] args) {
        // 创建队列（FIFO）
        Queue<Integer> queue = new ArrayDeque<>();

        // 1. 入队 add / offer
        queue.offer(10);  // 推荐
        queue.offer(20);
        queue.offer(30);
        System.out.println("队列内容: " + queue); // [10, 20, 30]

        // 2. 查看队头元素（不删除）
        System.out.println("队头元素 peek: " + queue.peek()); // 10

        // 3. 出队 poll（删除队头）
        int removed = queue.poll();
        System.out.println("出队元素: " + removed); // 10
        System.out.println("队列内容: " + queue);   // [20, 30]

        // 4. 判断是否为空
        System.out.println("队列是否为空: " + queue.isEmpty()); // false

        // 5. 队列大小
        System.out.println("队列大小: " + queue.size()); // 2

        // 6. 遍历队列（从队头到队尾）
        System.out.print("队列遍历: ");
        for (int num : queue) {
            System.out.print(num + " ");
        }
        System.out.println();
        
        // 7. 清空队列
        queue.clear();
        System.out.println("清空后是否为空: " + queue.isEmpty()); // true
    }
}
```
