## HashMap
```java
import java.util.HashMap;
import java.util.Map;

public class HashMapExample {
    public static void main(String[] args) {
        // 1. 创建 HashMap
        Map<String, Integer> map = new HashMap<>();

        // 2. 添加元素 put(key, value)
        map.put("Alice", 25);
        map.put("Bob", 30);
        map.put("Charlie", 35);

        // 3. 获取元素 get(key)
        System.out.println("Alice的年龄: " + map.get("Alice")); // 25

        // 4. 判断 key 是否存在 containsKey(key)
        System.out.println("是否包含Bob: " + map.containsKey("Bob")); // true

        // 5. 判断 value 是否存在 containsValue(value)
        System.out.println("是否有年龄为30的人: " + map.containsValue(30)); // true

        // 6. 删除元素 remove(key)
        map.remove("Charlie");
        System.out.println("删除Charlie后: " + map);

        // 7. 遍历 keySet
        System.out.println("遍历key:");
        for (String key : map.keySet()) {
            System.out.println(key);
        }

        // 8. 遍历 values
        System.out.println("遍历value:");
        for (Integer value : map.values()) {
            System.out.println(value);
        }

        // 9. 遍历 entrySet
        System.out.println("遍历键值对:");
        for (Map.Entry<String, Integer> entry : map.entrySet()) {
            System.out.println(entry.getKey() + " -> " + entry.getValue());
        }

        // 10. 获取大小 size
        System.out.println("Map大小: " + map.size());

        // 11. 清空 clear
        map.clear();
        System.out.println("清空后Map: " + map);
    }
}

```

```java
// 如果 key 已存在，返回已有的 value。

// 如果 key 不存在，使用 mappingFunction 生成 value 并插入 Map，然后返回该 value。

// 如果 mappingFunction 返回 null，不会插入任何值。

Map<String, List<String>> map = new HashMap<>();

// key "fruits" 不存在，会生成一个空列表并插入
List<String> fruits = map.computeIfAbsent("fruits", k -> new ArrayList<>());
fruits.add("apple");

// 再次调用不会创建新的列表，返回已有列表
map.computeIfAbsent("fruits", k -> new ArrayList<>()).add("banana");

System.out.println(map); // 输出：{fruits=[apple, banana]}
```

## HashSet
```java
import java.util.HashSet;
import java.util.Set;

public class HashSetExample {
    public static void main(String[] args) {
        // 1. 创建 HashSet
        Set<String> set = new HashSet<>();

        // 2. 添加元素 add(element)
        set.add("Apple");
        set.add("Banana");
        set.add("Orange");
        set.add("Apple"); // HashSet 不允许重复，重复添加无效

        System.out.println("删除Orange后: " + set);

            System.out.println(s);

        System.out.println("集合大小: " + set.size());

        // 7. 清空 clear()
        set.clear();
        System.out.println("清空后: " + set);
    }
}
```

## List
```java
import java.util.ArrayList;
import java.util.List;

public class ListExample {
    public static void main(String[] args) {
        // 1. 创建 List
        List<String> list = new ArrayList<>();

        // 2. 添加元素 add(element)
        list.add("Apple");
        list.add("Banana");
        list.add("Orange");

        // 3. 指定位置添加 add(index, element)
        list.add(1, "Grapes"); // 在索引1插入

        // 4. 获取元素 get(index)
        System.out.println("索引2的元素: " + list.get(2)); // Banana

        // 5. 修改元素 set(index, element)
        list.set(2, "Mango"); // 替换 Banana 为 Mango

        // 6. 删除元素 remove(index) 或 remove(object)
        list.remove(3); // 删除索引3的元素
        list.remove("Apple"); // 删除值为 Apple 的元素

        // 7. 判断是否包含 contains(element)
        System.out.println("是否包含Mango? " + list.contains("Mango"));

        // 8. 获取大小 size()
        System.out.println("List大小: " + list.size());

        // 9. 遍历元素
        System.out.println("遍历List:");
        for (String s : list) {
            System.out.println(s);
        }

        // 10. 清空 clear()
        list.clear();
        System.out.println("清空后: " + list);
    }
}
```

```java
public class StringDemo {

        String str = "Hello";

        String str2 = str + " World";

        char[] chars = str.toCharArray();
    }
}
```

```java
class Solution {
    int max = Integer.MIN_VALUE;
    public int maxPathSum(TreeNode root) {
        if(root == null) return 0;
        dfs(root);
        return max;
    }

    public int dfs(TreeNode root) {
        if(root == null) return 0;
        int lenum = Math.max(dfs(root.left), 0);
        int rinum = Math.max(dfs(root.right), 0);
        max= Math.max(max, lenum + rinum + root.val);
        return root.val + Math.max(lenum, rinum);
    }
}
```

## 链表
```java
class Solution {
    public ListNode reverseList(ListNode head) {
        ListNode next = head;			//主节点
        ListNode result = null;			//结果节点
        while(next != null) {
            ListNode temp = next.next;   //暂为主右，暂时节点是主节点下一个
            next.next = result;			//主向左，主节点指向左边的节点，也就是结果节点
            result = next;				//结向右
            next = temp;				//主向右
        }
        return result;
    }
}
```

```java
public class Solution {
    public boolean hasCycle(ListNode head) {
        if(head == null) return false;
        ListNode slow = head;
        ListNode fast = head;
        while(fast != null && fast.next != null) {	//注意条件
            slow = slow.next;
            fast = fast.next.next;
            if(slow == fast) return true;	//最后判断
        }
        return false;
    }
}
```

## 栈
```java
import java.util.Stack;

public class StackDemo {
    public static void main(String[] args) {
        // 创建栈
        Stack<Integer> stack = new Stack<>();

        // 1. 入栈 push
        stack.push(10);
        stack.push(20);
        stack.push(30);
        System.out.println("栈内容: " + stack); // 输出: [10, 20, 30]

        // 2. 查看栈顶元素 peek（不删除）
        System.out.println("栈顶元素: " + stack.peek()); // 输出: 30

        // 3. 出栈 pop（删除栈顶元素）
        int popped = stack.pop();
        System.out.println("弹出的元素: " + popped); // 输出: 30
        System.out.println("栈内容: " + stack);      // 输出: [10, 20]

        // 4. 判断栈是否为空 isEmpty
        System.out.println("栈是否为空: " + stack.isEmpty()); // 输出: false

        // 5. 栈大小 size
        System.out.println("栈大小: " + stack.size()); // 输出: 2

        // 6. 查找元素位置 search（返回从栈顶开始的距离，找不到返回 -1）
        System.out.println("20 在栈顶距离: " + stack.search(20)); // 输出: 1
        System.out.println("100 在栈顶距离: " + stack.search(100)); // 输出: -1

        // 遍历栈（从栈底到栈顶）
        System.out.print("栈遍历: ");
        for (int num : stack) {
            System.out.print(num + " ");
        }
        System.out.println();
    }
}
```
