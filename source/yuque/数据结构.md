## HashMap
```java
import java.util.HashMap;
import java.util.Map;

public class HashMapExample {
    public static void main(String[] args) {
        // 1. 创建 HashMap
        Map<String, Integer> map = new HashMap<>();

        // 2. 添加元素 put(key, value)
        map.put("Alice", 25);
        map.put("Bob", 30);
        map.put("Charlie", 35);

        // 3. 获取元素 get(key)
        System.out.println("Alice的年龄: " + map.get("Alice")); // 25

        // 4. 判断 key 是否存在 containsKey(key)
        System.out.println("是否包含Bob: " + map.containsKey("Bob")); // true

        // 5. 判断 value 是否存在 containsValue(value)
        System.out.println("是否有年龄为30的人: " + map.containsValue(30)); // true

        // 6. 删除元素 remove(key)
        map.remove("Charlie");
        System.out.println("删除Charlie后: " + map);

        // 7. 遍历 keySet
        System.out.println("遍历key:");
        for (String key : map.keySet()) {
            System.out.println(key);
        }

        // 8. 遍历 values
        System.out.println("遍历value:");
        for (Integer value : map.values()) {
            System.out.println(value);
        }

        // 9. 遍历 entrySet
        System.out.println("遍历键值对:");
        for (Map.Entry<String, Integer> entry : map.entrySet()) {
            System.out.println(entry.getKey() + " -> " + entry.getValue());
        }

        // 10. 获取大小 size
        System.out.println("Map大小: " + map.size());

        // 11. 清空 clear
        map.clear();
        System.out.println("清空后Map: " + map);
    }
}

```

## HashSet
```java
import java.util.HashSet;
import java.util.Set;

public class HashSetExample {
    public static void main(String[] args) {
        // 1. 创建 HashSet
        Set<String> set = new HashSet<>();

        // 2. 添加元素 add(element)
        set.add("Apple");
        set.add("Banana");
        set.add("Orange");
        set.add("Apple"); // HashSet 不允许重复，重复添加无效

        // 3. 判断是否包含 contains(element)
        System.out.println("包含Banana吗? " + set.contains("Banana")); // true

        // 4. 删除元素 remove(element)
        set.remove("Orange");
        System.out.println("删除Orange后: " + set);

        // 5. 遍历元素
        System.out.println("遍历元素:");
        for (String s : set) {
            System.out.println(s);
        }

        // 6. 获取大小 size()
        System.out.println("集合大小: " + set.size());

        // 7. 清空 clear()
        set.clear();
        System.out.println("清空后: " + set);
    }
}
```

## List
```java
import java.util.ArrayList;
import java.util.List;

public class ListExample {
    public static void main(String[] args) {
        // 1. 创建 List
        List<String> list = new ArrayList<>();

        // 2. 添加元素 add(element)
        list.add("Apple");
        list.add("Banana");
        list.add("Orange");

        // 3. 指定位置添加 add(index, element)
        list.add(1, "Grapes"); // 在索引1插入

        // 4. 获取元素 get(index)
        System.out.println("索引2的元素: " + list.get(2)); // Banana

        // 5. 修改元素 set(index, element)
        list.set(2, "Mango"); // 替换 Banana 为 Mango

        // 6. 删除元素 remove(index) 或 remove(object)
        list.remove(3); // 删除索引3的元素
        list.remove("Apple"); // 删除值为 Apple 的元素

        // 7. 判断是否包含 contains(element)
        System.out.println("是否包含Mango? " + list.contains("Mango"));

        // 8. 获取大小 size()
        System.out.println("List大小: " + list.size());

        // 9. 遍历元素
        System.out.println("遍历List:");
        for (String s : list) {
            System.out.println(s);
        }

        // 10. 清空 clear()
        list.clear();
        System.out.println("清空后: " + list);
    }
}
```
