```java
public static int binarySearch(int[] nums, int target) {
    int left = 0;
    int right = nums.length - 1;

    while (left <= right) {
        // 防止溢出写法
        int mid = left + (right - left) / 2;

        if (nums[mid] == target) {
            return mid;         // 找到
        } else if (nums[mid] < target) {
            left = mid + 1;    // target 在右半部分
        } else {
            right = mid - 1;   // target 在左半部分
        }
    }

    return -1;  // 不存在
    return l;  //返回应该添加到的位置
}
```

```java
public static int findKth(int[] a, int[] b, int k) {

    int i = 0;  // a 的指针
    int j = 0;  // b 的指针

    while (true) {

        // a 用完了，答案在 b
        if (i == a.length) {
            return b[j + k - 1];
        }

        // b 用完了，答案在 a
        if (j == b.length) {
            return a[i + k - 1];
        }

        // k == 1，只要比较当前两个最小的即可
        if (k == 1) {
            return Math.min(a[i], b[j]);
        }

        // 取 k/2 作为本轮要踢掉的数量
        int half = k / 2;

        // 找 a 的第 half 个候选位置（越界时设为无穷大）
        int newI = Math.min(i + half, a.length) - 1;
        int pivotA = a[newI];

        // 找 b 的第 half 个候选位置（越界时设为无穷大）
        int newJ = Math.min(j + half, b.length) - 1;
        int pivotB = b[newJ];

        // 谁小，谁那边的前 half 个都不可能是第 k 小
        if (pivotA <= pivotB) {
            k -= (newI - i + 1);  // 踢掉 a 的这些
            i = newI + 1;         // a 的指针向前走
        } else {
            k -= (newJ - j + 1);  // 踢掉 b 的这些
            j = newJ + 1;         // b 的指针向前走
        }
    }
}
```

```java
class Solution {
    public boolean searchMatrix(int[][] matrix, int target) {
        int m = matrix.length, n = matrix[0].length;
        int i = 0, j = n-1;
        while(i < m && j >= 0) {
            if(matrix[i][j] > target) j--;
            else if(matrix[i][j] < target) i++;
            else return true;
        } 
        return false;
    }
}
```

## 回溯算法
```java
class Solution {
    List<List<Integer>> res = new ArrayList<>();
    List<Integer> path = new ArrayList<>();
    boolean[] used;

    public List<List<Integer>> permute(int[] nums) {
        used = new boolean[nums.length];
        backtrack(nums);
        return res;
    }

    void backtrack(int[] nums) {
        if (path.size() == nums.length) {
            res.add(new ArrayList<>(path));
            return;
        }

        for (int i = 0; i < nums.length; i++) {
            if (used[i]) continue;   // 已经用过，跳过

            used[i] = true;          // 做选择
            path.add(nums[i]);

            backtrack(nums);         // 下一层继续选

            path.remove(path.size() - 1); // 撤销选择
            used[i] = false;
        }
    }
}
```

```java
class Solution {
    List<List<Integer>> res = new ArrayList<>();
    List<Integer> path = new ArrayList<>();

    public List<List<Integer>> combine(int n, int k) {
        backtrack(n, k, 1); // 从 1 开始选
        return res;
    }

    void backtrack(int n, int k, int start) {
        if (path.size() == k) {
            res.add(new ArrayList<>(path));
            return;
        }

        for (int i = start; i <= n; i++) {
            path.add(i);         // 选择
            backtrack(n, k, i+1);// 下一层从 i+1 开始
            path.remove(path.size()-1); // 撤销选择
        }
    }
}
```

```java
class Solution {
    List<List<Integer>> res = new ArrayList<>();
    List<Integer> path = new ArrayList<>();

    public List<List<Integer>> subsets(int[] nums) {
        backtrack(nums, 0);  // 从下标 0 开始
        return res;
    }

    void backtrack(int[] nums, int start) {
        res.add(new ArrayList<>(path));  // 每一条路径都是一个子集

        for (int i = start; i < nums.length; i++) {
            path.add(nums[i]);         // 做选择
            backtrack(nums, i + 1);    // 递归，下一层从 i+1 开始
            path.remove(path.size() - 1); // 撤销选择
        }
    }
}
```
